test('POST - Successful login @master @sanity', async ({ page }) => {
  // Use page.evaluate() to run the script inside a genuine browser context
  const responseData = await page.evaluate(async (loginUrl) => {
    const res = await fetch(loginUrl, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      },
      body: JSON.stringify({ 
        username: 'mor_2314', 
        password: '83r5^_' 
      })
    });

    // Capture both the HTTP status and the body text safely
    return {
      status: res.status,
      text: await res.text()
    };
  }, endpoint(Routes.AUTH_LOGIN));

  // Log information out to your console for immediate verification
  console.log("Cloudflare-Bypass Response Status:", responseData.status);
  console.log("Cloudflare-Bypass Response Body:", responseData.text);

  // 1. Assert our clean 200 validation
  expect(responseData.status).toBe(200);

  // 2. Assert the authentication token structural rules
  const body = JSON.parse(responseData.text);
  expect(body.token).toEqual(expect.any(String));
  expect(body.token.length).toBeGreaterThan(0);
});
